// Initialize with .manual to skip auto-init
$('[data-sort-id="gallery4"]').closest('.sort-source')
    // or just call manually:
$('[data-plugin-sort][data-sort-id="gallery4"]').themestrapPluginSort({
    useHash: false
});

// Get instance and drive it
const instance = $('[data-plugin-sort][data-sort-id="gallery4"]').data('__sort');
instance.setFilter('.tier-gold');
instance.setFilter('*'); // show all

Options Reference

Option Type Default Description
useHash Boolean true Sync active filter to URL hash and react to hashchange events.
itemSelector String ".isotope-item" CSS selector for grid items inside the destination element.
layoutMode String "masonry" "masonry" (native absolute positioning) or "fitRows" (CSS flex wrap).
filter String "*" Initial active filter. Use "*" for all or a CSS class selector like ".cat-nature".
filterFieldId String / false false ID of an <input> element whose value is used for live text search against data-sort-search.
filterFieldText String "" Internal: current value of the search field. Not typically set manually.
hiddenStyle Object { opacity: 0 } CSS properties applied to filtered-out items.
visibleStyle Object { opacity: 1 } CSS properties applied to visible items.
stagger Number 30 Per-item transition-delay increment in ms for masonry layout (staggered cascade effect).
isOriginLeft Boolean true Set false for RTL layouts. Auto-detected from <html dir="rtl">.
