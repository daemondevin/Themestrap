Programmatic API _setPosition · destroy · reinit

Position

Lifecycle

Current pct

0.5000
// Access instance
const ba = $('#ba-api').data('__beforeafter');

// Jump to any position (0.0 – 1.0)
ba._setPosition(0.25);

// Read current position
console.log(ba.pct); // 0.25

// Destroy (restores original markup)
ba.destroy();

// Reinit with new options
$('#ba-api').themestrapPluginBeforeAfter({ default_offset_pct: 0.75 });
