Programmatic API — next / prev / to / destroy

// .manual skips auto-init; call themestrapPluginCarousel() yourself
const $el   = $('#api-carousel').themestrapPluginCarousel();
const inst  = $el.data('__carousel');

inst.prev();
inst.next();
inst.to(2);          // zero-based index
inst.destroy();     // removes clones, unbinds, clears data

$el.on('changed.owl.carousel', e => {
  console.log('slide', e.item.index);
});
// click a button above — changed.owl.carousel events logged here
