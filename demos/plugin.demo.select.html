PluginSelect

A fully accessible custom select that wraps a native <select> for form compatibility, supports option groups, keyboard navigation, type-ahead, and flip-up positioning.

01 Basic Select Single select 
/* wrap any <select> — plugin reads options automatically */
<div data-plugin-select>
  <select name="timezone">
    <option value="">Select a timezone…</option>
    <option value="utc">UTC</option>
    <option value="est">Eastern (EST / EDT)</option>
    …
  </select>
</div>
02 Option Groups & States Groups · Disabled · Pre-selected 
/* option groups via <optgroup> */
<optgroup label="Frontend">
  <option value="react">React</option>
</optgroup>

/* disabled trigger — set disabled on the <select> */
<select name="plan" disabled>…</select>

/* disabled individual options */
<option value="lg" disabled>LG (out of stock)</option>
03 Dark Mode data-bs-theme 
/* dark mode activates via Bootstrap or the html.dark class */
<div data-bs-theme="dark">
  <div data-plugin-select>…</div>
</div>

/* retrieve instance */
const sel = $('#api-select').data('__pluginSelect');

sel.open();            /* open the dropdown */
sel.close();           /* close the dropdown */
sel.select('apple');  /* select by value */
sel.setValue('apple'); /* alias for select() */
sel.getValue();        /* returns current value string */
sel.reset();           /* clear selection, restore placeholder */
sel.enable();
sel.disable();
sel.destroy();         /* teardown, restores original HTML */

/* events fire on the wrapper element and bubble */

    console.log(e.detail); /* { value, text, previousValue } */
});
