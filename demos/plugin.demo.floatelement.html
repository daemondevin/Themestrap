// Normal horizontal
$('#h1').themestrapPluginFloatElement({
  horizontal: true,
  speed:      4,
});

// invertDirection flips the sign — replaces the old minus: '-' pattern
$('#h2').themestrapPluginFloatElement({
  horizontal:      true,
  speed:           4,
  invertDirection: true,
});
// top pill and bottom pill slide in opposite directions

// Each transform takes a [from, to] tuple, interpolated 0–1
// across the normalised scroll position of the element.
$('#hero-card').themestrapPluginFloatElement({
  speed:              4,
  scale:              [1, 1.07],    // grows as element scrolls up
  rotate:             [-4, 4],      // degrees
  transition:         true,
  transitionDuration: 200,          // ms — absorbs micro-jitter
});

$('#orb').themestrapPluginFloatElement({
  speed:   3,
  opacity: [0.4, 1],  // fades in as it scrolls into view
  scale:   [0.9, 1.1],
});

$('#panel').themestrapPluginFloatElement({
  speed:   5,
  skew:    [0, 6],    // skewX in degrees
  opacity: [1, 0.4],
});
// scroll to see scale, rotate, skew, and opacity animate simultaneously

↕ Scroll down then back up — watch the directional orbs freeze
// clamp: N caps the translate at ±N% regardless of scroll distance
$('#constrained').themestrapPluginFloatElement({
  speed: 2,
  clamp: 4,
});

// scrollDirection: 'down' — transform only updates while scrolling down
// scrollDirection: 'up'   — transform only updates while scrolling up
// scrollDirection: 'both' — always updates (default)
$('#down-only').themestrapPluginFloatElement({
  speed:           3,
  scrollDirection: 'down',
});
// scroll down then back up — watch the right two orbs freeze mid-drift

state: active
↕ Scroll or use the buttons below to control the instance
// Instance retrieval (never chain off the init call)
const inst = $('#fe').data('__floatElement');

inst.pause();        // suspend rAF updates
inst.resume();       // resume + immediately recalculates
inst.recalculate();  // force one tick without scrolling
inst.destroy();      // unbinds scroll, clears transform, disconnects IO

// String command shorthand via the jQuery bridge:
$('#fe').themestrapPluginFloatElement('pause');

// onMove fires on every computed rAF tick
$('#fe').themestrapPluginFloatElement({
  speed:   3,
  scale:   [1, 1.1],
  opacity: [0.6, 1],
  onMove: (pct, transforms) => {
    console.log(pct, transforms.translateY, transforms.scale);
  },
});
// scroll or click a button above
