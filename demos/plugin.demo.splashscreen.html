Splash Screen

See PluginSplashScreen in action. Each demo mounts the overlay on a contained #demo-overlay element, downloads a small set of real CDN assets, then dismisses itself. The demo log tracks every event so you can follow the load sequence in real time.

$('#overlay-basic').themestrapPluginSplashScreen({
    files:   DEMO_FILES,
    threads: 1,
    debug:   true,
    onLoaded(current, total, path) {
        const name = path.split('/').pop();
        demoLog(log, `(${current}/${total}) ${name}`, 'log-info');
    },
    onSuccess() {
        demoLog(log, 'onSuccess — all files injected', 'log-ok');
        document.getElementById('ready-basic').classList.add('visible');
        setTimeout(() => {
            document.getElementById('overlay-basic').classList.add('hidden');
        }, 1200);
    }
});

$('#overlay-progress').themestrapPluginSplashScreen({
    files:   DEMO_FILES,
    threads: 1,
    onLoaded(current, total, path) {
        const name = path.split('/').pop();
        document.getElementById('prog-title').textContent =
            `Downloading ${current} of ${total}`;
        document.getElementById('prog-fill').style.width =
            Math.round((current / total) * 100) + '%';
        document.getElementById('prog-file').textContent = name;
    },
    onSuccess() {
        document.getElementById('prog-title').textContent = 'Complete!';
        document.getElementById('prog-fill').style.width = '100%';
        document.getElementById('prog-file').textContent = '';
        document.getElementById('ready-progress').classList.add('visible');
        setTimeout(() => {
            document.getElementById('overlay-progress').classList.add('hidden');
        }, 1400);
    }
});

$('#overlay-ribbon').themestrapPluginSplashScreen({
    files:   DEMO_FILES,
    threads: 1,
    ribbon:  true,            // lazy-loads THREE.js r128 from jsDelivr
    debug:   true,
    onLoaded(current, total, path) {
        document.getElementById('ribbon-fill').style.width =
            Math.round((current / total) * 100) + '%';
        document.getElementById('ribbon-file').textContent =
            path.split('/').pop();
    },
    onSuccess() {
        document.getElementById('ribbon-title').textContent = 'Done';
        document.getElementById('ready-ribbon').classList.add('visible');
        /* ribbon keeps animating until destroy() */
    }
});
 
/* Stop ribbon and clean up: */
$('#overlay-ribbon').data('__splashScreen').destroy();

getFile()

Downloads a small JSON file, reads it back via getFile(), and displays the parsed result.

Fetching config…
— ready to run —
Show code 
/* Download a JSON file and read its content after load */
const CONFIG_URL = 'https://jsonplaceholder.typicode.com/todos/1';
 
const splash = $('#overlay-getfile').themestrapPluginSplashScreen({
    files:   [CONFIG_URL],
    threads: 1,
    fresh:   true,           // always fetch — don't cache placeholder data
    onSuccess() {
        const raw    = splash.getFile(CONFIG_URL);
        const parsed = JSON.parse(raw);
 
        document.getElementById('overlay-getfile').classList.add('hidden');
        const out = document.getElementById('getfile-result');
        out.textContent = JSON.stringify(parsed, null, 2);
        out.style.display = 'block';
    }
});

/* 1. Initialize PluginProgressBar first so its instance is in jQuery data */
$('#pb-bar').themestrapPluginProgressBar({
    value:    0,
    color:    'success',
    size:     'lg',
    label:    true,
    duration: 400,
});
 
/* 2. Listen to native events fired by PluginProgressBar */
document.getElementById('pb-bar').addEventListener('progressbar:change', (e) => {
    console.log('bar changed →', e.detail.value);
});
document.getElementById('pb-bar').addEventListener('progressbar:complete', () => {
    console.log('bar reached 100%');
});
 
/* 3. Pass the selector to SplashScreen — set() is called automatically */
$('#overlay-pb').themestrapPluginSplashScreen({
    files:       DEMO_FILES,
    threads:     1,
    progressBar: '#pb-bar',   // SplashScreen drives the bar; no onLoaded needed
    onLoaded(current, total, path) {
        /* onLoaded still fires — use it for text readouts */
        document.getElementById('pb-title').textContent =
            'Downloading ' + current + ' of ' + total;
        document.getElementById('pb-file').textContent =
            path.split('/').pop();
    },
    onSuccess() {
        document.getElementById('pb-title').textContent = 'Complete!';
        document.getElementById('pb-file').textContent  = '';
        document.getElementById('ready-pb').classList.add('visible');
        setTimeout(() => {
            document.getElementById('overlay-pb').classList.add('hidden');
        }, 1400);
    }
});
