const throttle = (fn, delay) => {
    let last = 0;
    return (...args) => {
        const now = Date.now();
        if (now - last >= delay) {
            last = now;
            return fn(...args);
        }
    };
};

class EventEmitter {
    constructor() {
        this._events = Object.create(null);
    }

    on(event, fn) {
        (this._events[event] ??= []).push(fn);
        return this;
    }

    emit(event, ...args) {
        (this._events[event] || []).forEach(fn => fn(...args));
        return this;
    }
}

const bus = new EventEmitter();
bus.on('ready', () => console.log('Plugin ready.'));
bus.emit('ready');

PluginHighlight v2.0.0
======================
Wraps highlight.js with:
  * Lazy ESM loading from CDN
  * Retry + backoff on CDN failure
  * Line numbers with anchor IDs
  * Click-drag line selection
  * Clipboard copy (modern + fallback)
  * 31 CSS custom properties
  * MODX language support

Omit data-plugin-highlight or leave it blank
for plaintext — line numbers and copy still work.
/* Auto-init (via themestrap.init.js) */
if ($.isFunction($.fn['themestrapPluginHighlight']) && $('[data-plugin-highlight]').length) {
    themestrap.fn.intObsInit('[data-plugin-highlight]:not(.manual)', 'themestrapPluginHighlight');
}

/* Manual init */
$('#ex-js').themestrapPluginHighlight();

Language is set via data-plugin-highlight. Aliases: js → javascript, html → xml, sh → bash, yml → yaml.
