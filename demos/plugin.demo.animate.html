  $(this).themestrapPluginAnimate({ forceAnimation: true });
});

03

Flag-class-only — let CSS handle the animation

flagClassOnly: true
Item A — slides in from below, driven by CSS transition on .appear-animation-visible
Item B — same transition, 200 ms delay on CSS transition-delay
Item C — 400 ms transition-delay 
/* CSS owns the animation — plugin only adds the class */
.my-item {
  transform: translateY(20px); opacity: 0;
  transition: transform .5s ease, opacity .5s ease;
}
.my-item.appear-animation-visible {
  transform: translateY(0);  opacity: 1;
}

$('.my-item').themestrapPluginAnimate({
  flagClassOnly:  true,  // skip Animate.css class; just add .appear-animation-visible
  forceAnimation: true,
});
// plugin adds .appear-animation-visible; CSS does the rest
